\section{Vì sao AI cần NumPy?}

\begin{frame}{Dữ liệu trong AI là những con số --- rất nhiều con số}
\begin{center}
  \includegraphics[width=0.92\textwidth]{figures/imagematrix.png}
\end{center}
\begin{itemize}
  \item Máy tính \textbf{không "nhìn thấy" ảnh} --- nó chỉ thấy một \textbf{ma trận các con số},
        mỗi số là độ sáng của một điểm ảnh (pixel), từ 0 (đen) đến 255 (trắng)
  \item Chỉ một ảnh bé xíu như trên đã là hàng trăm con số. Một tấm ảnh
        $224 \times 224$ màu $\to$ $224 \times 224 \times 3 = 150\,528$ con số
  \item Bảng điểm 1000 học sinh $\times$ 10 môn $\to$ $10\,000$ con số
\end{itemize}
\end{frame}

\begin{frame}{Vậy xử lý ngần ấy con số bằng gì?}
\begin{block}{Câu hỏi}
Nếu dùng \texttt{list} của Python thuần và vòng lặp \texttt{for} để xử lý từng con số một, chuyện gì sẽ xảy ra?
\end{block}
\pause
\begin{alertblock}{Trả lời}
Chương trình chạy \textbf{rất chậm}. Với hàng triệu con số, thứ đáng lẽ chạy trong vài giây có thể mất hàng phút, thậm chí hàng giờ.
\end{alertblock}
\pause
\begin{block}{Lời giải}
Ta cần một công cụ chuyên để chứa và tính toán trên \textbf{cả một khối số cùng lúc} --- đó chính là \textbf{NumPy}.
\end{block}
\end{frame}

\begin{frame}{NumPy là gì?}
\begin{itemize}
  \item \textbf{NumPy} (Numerical Python) --- thư viện tính toán số học nền tảng của Python
  \item Ra đời năm 2006, hiện là "trái tim" của toàn bộ hệ sinh thái khoa học dữ liệu và AI
  \item Cung cấp kiểu dữ liệu \textbf{\texttt{ndarray}} --- mảng nhiều chiều, lưu số hiệu quả
  \item Phần lõi được viết bằng \textbf{C} và Fortran $\to$ nhanh hơn Python thuần hàng chục đến hàng trăm lần
\end{itemize}
\pause
\begin{block}{Ai đang dùng NumPy?}
Pandas, Scikit-learn, Matplotlib, OpenCV, PyTorch, TensorFlow --- tất cả đều xây dựng trên NumPy hoặc dùng chung cách tổ chức dữ liệu của NumPy.
\end{block}
\end{frame}

\begin{frame}[fragile]{So sánh tốc độ: list vs ndarray}
\begin{lstlisting}
import numpy as np, time

n = 1_000_000
list_a = list(range(n))
arr_a  = np.arange(n)

t0 = time.time()
tong1 = sum(x * x for x in list_a)   # Python thuan
t1 = time.time()
tong2 = np.sum(arr_a * arr_a)        # NumPy
t2 = time.time()

print("Python list:", t1 - t0, "giay")
print("NumPy      :", t2 - t1, "giay")
\end{lstlisting}
\begin{block}{Kết quả tham khảo (tùy máy)}
Python list $\approx$ 0.025--0.09 giây; NumPy $\approx$ 0.001--0.002 giây
$\to$ \textbf{nhanh hơn 20--50 lần}
\end{block}
\end{frame}

\begin{frame}{NumPy array khác Python list ở điểm nào?}
\begin{center}
\small
\begin{tabular}{|l|l|l|}
\hline
 & \textbf{Python list} & \textbf{NumPy ndarray} \\
\hline
Kiểu phần tử      & Trộn lẫn tùy ý       & \textbf{Đồng nhất} (int, float, \ldots) \\
Kích thước        & Thay đổi được        & Cố định khi tạo \\
Bộ nhớ            & Nhiều, rải rác       & Ít, liên tục \\
Tốc độ tính toán  & Chậm (vòng lặp)      & Nhanh (chạy trong C) \\
Phép toán số học  & \texttt{[1,2]+[3,4]} $\to$ nối & \texttt{a+b} $\to$ cộng từng phần tử \\
Nhiều chiều       & Phải lồng list       & Hỗ trợ sẵn \\
Hàm toán học      & Phải tự viết         & Có sẵn hàng trăm hàm \\
\hline
\end{tabular}
\end{center}
\begin{alertblock}{Ghi nhớ}
List dùng để chứa \textbf{mọi thứ}. NumPy array dùng để \textbf{tính toán trên số}.
\end{alertblock}

\section{Bắt đầu với NumPy}

\begin{frame}[fragile]{Cài đặt và import}
\textbf{Cài đặt} (chỉ làm một lần, trong Terminal / Command Prompt):
\begin{lstlisting}[language=bash, numbers=none, keywordstyle=\color{black}]
pip install numpy
\end{lstlisting}
Với Google Colab hoặc Anaconda: NumPy đã được cài sẵn.
\vspace{0.3cm}
\textbf{Import} --- quy ước chung của cả thế giới:
\begin{lstlisting}
import numpy as np

print(np.__version__)   # kiem tra phien ban
\end{lstlisting}
\begin{block}{\texttt{as np} nghĩa là gì?}
\texttt{as} đặt cho thư viện một \textbf{tên gọi tắt} (tiếng Anh: \textit{alias}).
Sau dòng đó, thay vì viết \texttt{numpy.array(...)} ta chỉ cần viết \texttt{np.array(...)} --- ngắn hơn, gõ nhanh hơn.
Cả thế giới đều dùng tên tắt \texttt{np}, nên hãy viết y hệt vậy để đọc code của người khác không bị lạ.
\end{block}
\end{frame}

\begin{frame}[fragile]{Mảng đầu tiên của bạn}
\begin{lstlisting}
import numpy as np

a = np.array([1, 2, 3, 4, 5])
print(a)          # [1 2 3 4 5]
print(type(a))    # <class 'numpy.ndarray'>

# Phep toan tac dong len TOAN BO mang
print(a + 10)     # [11 12 13 14 15]
print(a * 2)      # [ 2  4  6  8 10]
print(a ** 2)     # [ 1  4  9 16 25]
\end{lstlisting}
\begin{itemize}
  \item Không cần vòng lặp \texttt{for} --- đây gọi là \textbf{vector hóa} (vectorization)
  \item So sánh: với list, \texttt{[1,2,3] * 2} cho \texttt{[1,2,3,1,2,3]} --- hoàn toàn khác!
\end{itemize}
\end{frame}

\begin{block}{Thuật ngữ: \textbf{tensor}}
Trong AI, mảng nhiều chiều được gọi chung là \textbf{tensor}: một số lẻ là tensor 0 chiều,
vector là 1 chiều, ma trận là 2 chiều, từ 3 chiều trở lên vẫn gọi là tensor.
Mọi dữ liệu --- ảnh, âm thanh, văn bản, bảng số --- đều được biểu diễn bằng tensor.
\end{block}

\begin{frame}[fragile]{Tạo mảng 2 chiều}

diem = np.array([[8.0, 7.5, 9.0],      # hoc sinh 1
                 [6.5, 8.0, 7.0],      # hoc sinh 2

  \item Mỗi list con là một \textbf{hàng} (row) và phải có \textbf{cùng độ dài}
  \item Quy ước đọc \texttt{shape}: (số hàng, số cột)
\end{itemize}
\end{frame}

\section{Các cách tạo mảng}

\begin{frame}[fragile]{Tạo mảng từ giá trị có sẵn}
\begin{lstlisting}
np.zeros(shape, dtype)   # CU PHAP: hinh dang, roi kieu du lieu

np.zeros(5)          # shape = 5       -> [0. 0. 0. 0. 0.]
np.zeros((2, 3))     # shape = (2, 3)  -> ma tran 2x3 toan so 0
np.ones((2, 3))      # ma tran 2x3 toan so 1
np.full((2, 3), 7)   # (shape, gia tri) -> 2x3 toan so 7
np.eye(3)            # ma tran don vi 3x3 (duong cheo = 1)
\end{lstlisting}
\begin{itemize}
  \item Tham số \textbf{thứ nhất} là \texttt{shape} --- hình dạng. Nhiều chiều thì phải gói tất cả
        vào \textbf{một} cặp ngoặc: \texttt{(2, 3)}. Đó là lý do \texttt{np.zeros((2, 3))} có \textbf{hai} lớp ngoặc
  \item Tham số \textbf{thứ hai} là \texttt{dtype} --- kiểu dữ liệu, \textbf{không phải} số cột
\end{itemize}
\begin{alertblock}{Vì vậy \texttt{np.zeros(2, 3)} sẽ báo lỗi}
NumPy hiểu thành \texttt{shape = 2}, \texttt{dtype = 3} --- mà 3 không phải là một kiểu dữ liệu.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{Tạo dãy số: arange và linspace}
\begin{lstlisting}
np.arange(start, stop, step)   # dem tu start, moi lan +step
np.linspace(start, stop, num)  # chia deu doan [start, stop]

np.arange(5)             # [0 1 2 3 4]        stop = 5
np.arange(2, 10)         # [2 3 4 5 6 7 8 9]  start=2, stop=10
np.arange(0, 10, 2)      # [0 2 4 6 8]        step = 2
np.linspace(0, 1, 5)     # [0. 0.25 0.5 0.75 1.]   num = 5 so
\end{lstlisting}
\begin{center}
\small
\begin{tabular}{|l|l|}
\hline
\textbf{Tham số} & \textbf{Ý nghĩa} \\
\hline
\texttt{start} & giá trị bắt đầu --- luôn được lấy \\
\texttt{stop}  & giá trị dừng: \texttt{arange} \textbf{không} lấy, \texttt{linspace} \textbf{có} lấy \\
\texttt{step}  & bước nhảy --- chỉ \texttt{arange} có \\
\texttt{num}   & số phần tử --- chỉ \texttt{linspace} có \\
\hline
\end{tabular}
\end{center}
\begin{itemize}
  \item Vẽ đồ thị hàm số $\to$ dùng \texttt{linspace}; tạo chỉ số đếm $\to$ dùng \texttt{arange}
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Tạo mảng ngẫu nhiên}
\begin{lstlisting}
rng = np.random.default_rng(42)   # 42 la "hat giong" (seed)

rng.random(5)             # 5 so thuc ngau nhien trong [0, 1)
rng.random((2, 3))        # ma tran 2x3 so thuc ngau nhien
rng.integers(1, 7, 10)    # 10 so nguyen 1..6 (gieo xuc xac)
rng.normal(0, 1, 5)      # 5 so phan phoi chuan (mean 0, std 1)
\end{lstlisting}
\begin{block}{Vì sao cần seed?}
Đặt seed $\to$ lần chạy nào cũng cho \textbf{cùng một kết quả} $\to$ thí nghiệm AI có thể \textbf{lặp lại được} (reproducible). Đây là nguyên tắc bắt buộc khi làm nghiên cứu.
\end{block}
\begin{itemize}
  \item Cách cũ (vẫn gặp nhiều trong tài liệu): \texttt{np.random.seed(42)} rồi \texttt{np.random.rand(5)}
\end{itemize}
\end{frame}

\section{Thuộc tính và kiểu dữ liệu}

\begin{frame}[fragile]{Các thuộc tính quan trọng của mảng}
\begin{lstlisting}
a = np.array([[1, 2, 3],
              [4, 5, 6]])
\end{lstlisting}
\begin{center}
\small
\begin{tabular}{|l|l|l|}
\hline
\textbf{Thuộc tính} & \textbf{Ý nghĩa} & \textbf{Với \texttt{a} ở trên} \\
\hline
\texttt{a.shape}    & hình dạng: (số hàng, số cột) & \texttt{(2, 3)} \\
\texttt{a.ndim}     & số chiều                     & \texttt{2} \\
\texttt{a.size}     & tổng số phần tử              & \texttt{6} \\
\texttt{a.dtype}    & kiểu dữ liệu của phần tử     & \texttt{int64} \\
\texttt{a.itemsize} & số byte của mỗi phần tử      & \texttt{8} \\
\texttt{a.nbytes}   & tổng số byte (size $\times$ itemsize) & \texttt{48} \\
\hline
\end{tabular}
\end{center}
\begin{itemize}
  \item Đây là \textbf{thuộc tính}, không phải hàm $\to$ viết \texttt{a.shape}, \textbf{không} có dấu ngoặc
\end{itemize}
\begin{alertblock}{Thói quen tốt số 1}
Gặp lỗi hay nghi ngờ $\to$ \texttt{print(x.shape)} \textbf{ngay lập tức}. Phần lớn lỗi khi làm việc với mảng đến từ \textbf{sai hình dạng}.
\end{alertblock}
\end{frame}

\begin{frame}[fragile]{Kiểu dữ liệu (dtype)}
\begin{lstlisting}
np.array([1, 2, 3]).dtype         # int64  (Windows cu: int32)
np.array([1.0, 2.0]).dtype         # float64
np.array([1, 2], dtype=np.float32) # ep kieu ngay khi tao

a = np.array([1.7, 2.3, 3.9])
b = a.astype(int)      # [1 2 3] - CAT phan thap phan
c = a.astype(np.float32)
\end{lstlisting}
\begin{center}
\small
\begin{tabular}{|l|l|l|}
\hline
\textbf{Kiểu} & \textbf{Bộ nhớ} & \textbf{Dùng khi} \\
\hline
\texttt{uint8}   & 1 byte & Ảnh (giá trị 0--255) \\
\texttt{int32/int64} & 4/8 byte & Nhãn, chỉ số, số đếm \\
\texttt{float32} & 4 byte & Số thực, tiết kiệm bộ nhớ \\
\texttt{float64} & 8 byte & Mặc định của NumPy, độ chính xác cao \\
\texttt{bool}    & 1 byte & Mặt nạ lọc dữ liệu \\
\hline
\end{tabular}
\end{center}
\end{frame}

\begin{frame}[fragile]{Cẩn thận với kiểu số nguyên}
\begin{lstlisting}
a = np.array([250, 251, 252], dtype=np.uint8)
print(a + 10)      # [4 5 6]  ??? -- TRAN SO (overflow)!
# uint8 chi chua duoc 0..255, 260 quay vong ve 4

b = np.array([1, 2, 3])
b[0] = 3.9
print(b)           # [3 2 3]  -- bi cat thanh so nguyen
\end{lstlisting}
\begin{alertblock}{Quy tắc an toàn khi làm AI}
Trước khi tính toán trên ảnh, luôn đổi sang số thực:
\texttt{img = img.astype(np.float32) / 255.0}
\end{alertblock}

\section{Truy cập phần tử: indexing và slicing}

\begin{frame}[fragile]{Indexing và slicing mảng 1 chiều}
\begin{lstlisting}
a = np.array([10, 20, 30, 40, 50, 60])

a[0]        # 10   phan tu dau tien
a[-1]       # 60   phan tu cuoi cung
a[1:4]      # [20 30 40]   tu chi so 1 den 3

a[3:]       # [40 50 60]   tu chi so 3 den het
a[::2]      # [10 30 50]   cach 1 lay 1
a[::-1]     # [60 50 40 30 20 10]   dao nguoc
\end{lstlisting}
\begin{itemize}
  \item Cú pháp giống hệt list của Python: \texttt{a[start : stop : step]} --- bắt đầu, kết thúc, bước nhảy
  \item Luôn nhớ: chỉ số bắt đầu từ \textbf{0}, và \textbf{không} lấy phần tử ở vị trí \texttt{stop}
\end{itemize}

\begin{frame}[fragile]{Lọc dữ liệu bằng điều kiện (boolean masking)}
\begin{lstlisting}
diem = np.array([8.5, 4.0, 7.0, 9.5, 5.5, 3.0])

mask = diem >= 5
print(mask)          # [True False True True True False]
print(diem[mask])    # [8.5 7.  9.5 5.5]   chi lay diem >= 5
print(diem[diem >= 5])            # viet gon, rat pho bien

print(np.sum(diem >= 5))          # 4  - dem so hoc sinh dat
print(np.mean(diem >= 5))         # 0.666... - ti le dat

# Ket hop nhieu dieu kien: dung & (va), | (hoac), ~ (phu dinh)
print(diem[(diem >= 5) & (diem < 9)])   # [8.5 7.  5.5]
\end{lstlisting}
\begin{alertblock}{Chú ý}
Dùng \texttt{\&}, \texttt{|} chứ \textbf{không} dùng \texttt{and}, \texttt{or}. Và \textbf{bắt buộc} có dấu ngoặc quanh mỗi điều kiện.
\end{alertblock}

\end{lstlisting}

\begin{lstlisting}
diem = np.array([8.5, 4.0, 7.0, 3.0])

np.where(diem >= 5, 1, 0)      # [1 0 1 0] - gan nhan Dat/Truot
np.where(diem >= 5, diem, 0)    # [8.5 0. 7. 0.] - truot -> 0
np.where(diem >= 5)             # (array([0, 2]),) - vi tri
\end{lstlisting}
\begin{itemize}
  \item \texttt{np.where} chính là cách viết \texttt{if--else} \textbf{cho cả mảng}, không cần vòng lặp
\end{itemize}
\end{frame}

\section{View và Copy --- cái bẫy kinh điển}

\begin{frame}[fragile]{Slicing tạo ra "view", không phải bản sao}
\begin{lstlisting}
a = np.array([1, 2, 3, 4, 5])
b = a[1:4]       # b la VIEW - nhin vao cung vung nho voi a
b[0] = 99

print(b)          # [99  3  4]
print(a)          # [ 1 99  3  4  5]   <-- a CUNG BI DOI!
\end{lstlisting}
\pause
\textbf{Muốn có bản sao độc lập $\to$ dùng \texttt{.copy()}:}
\begin{lstlisting}
c = a[1:4].copy()
c[0] = 0
print(a)          # [ 1 99  3  4  5]   <-- a khong doi
\end{lstlisting}
\begin{block}{Vì sao NumPy làm vậy?}
Để \textbf{tiết kiệm bộ nhớ} --- cắt một mảng 1GB mà phải sao chép thì rất tốn kém.
\end{block}
\end{frame}

\begin{frame}[fragile]{Khi nào là view, khi nào là copy?}
\begin{center}
\small
\begin{tabular}{|l|l|}
\hline
\textbf{Thao tác} & \textbf{Kết quả} \\
\hline
\texttt{b = a[1:4]}          & \textbf{View} --- dùng chung bộ nhớ \\
\texttt{b = a[a > 5]}        & Copy (boolean masking) \\
\texttt{b = a[[0, 2]]}       & Copy (fancy indexing) \\
\texttt{b = a.reshape(2, 3)} & \textbf{View} (nếu có thể) \\
\texttt{b = a.T}             & \textbf{View} \\
\texttt{b = a.copy()}        & Copy \\
\texttt{b = a + 0}           & Copy (mảng mới từ phép toán) \\
\hline
\end{tabular}
\end{center}
\begin{lstlisting}
b = a[1:4]
print(b.base is a)   # True  -> b la view cua a
\end{lstlisting}
\begin{alertblock}{Quy tắc vàng cho người mới}
Nếu bạn định \textbf{sửa} dữ liệu và \textbf{không} muốn ảnh hưởng mảng gốc $\to$ luôn gọi \texttt{.copy()}.
\end{alertblock}
\end{frame}

\section{Vector hóa --- tính toán không cần vòng lặp}

\begin{frame}[fragile]{Phép toán trên từng phần tử (element-wise)}
\begin{lstlisting}
a = np.array([1, 2, 3, 4])
b = np.array([10, 20, 30, 40])

a + b       # [11 22 33 44]
b - a       # [ 9 18 27 36]
a * b       # [ 10  40  90 160]  nhan TUNG PHAN TU
b / a       # [10. 10. 10. 10.]
a ** 2      # [ 1  4  9 16]
b % 3       # [1 2 0 1]

a > 2       # [False False  True  True]
\end{lstlisting}
\begin{itemize}
  \item Hai mảng phải có \textbf{cùng hình dạng} (hoặc tuân theo quy tắc broadcasting ở phần sau)
  \item Kết quả luôn là một mảng \textbf{mới}, cùng hình dạng
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Các hàm toán học có sẵn (universal functions)}
\begin{lstlisting}
a = np.array([1, 4, 9, 16])

np.sqrt(a)      # [1. 2. 3. 4.]      can bac hai
np.exp(a)       # e^a                 ham mu
np.log(a)       # ln(a)               logarit tu nhien
np.abs(-a)      # gia tri tuyet doi
np.round(np.array([1.234, 5.678]), 1)   # [1.2 5.7]

x = np.linspace(0, np.pi, 5)
np.sin(x)       # ham luong giac
np.maximum(a, 5)   # [ 5  5  9 16]  so sanh tung phan tu voi 5
\end{lstlisting}
\begin{block}{Vì sao quan trọng với AI?}
Các hàm hay gặp trong AI --- sigmoid, tanh, ReLU, softmax (sẽ gặp ở cuối buổi) --- đều được viết chỉ bằng vài hàm trên.
\end{block}
\end{frame}

\begin{frame}[fragile]{Vector hóa: viết lại vòng lặp cho gọn và nhanh}
\begin{block}{Bài toán}
Mảng \texttt{diem} chứa điểm của cả lớp. Cần quy đổi sang thang mới theo công thức
\textbf{điểm mới $=$ điểm cũ $\times 2 + 1$} cho \textbf{mọi} học sinh.
\end{block}
\textbf{Cách viết của người mới --- duyệt từng học sinh một:}
\begin{lstlisting}
ket_qua = []                        # danh sach rong
for i in range(len(diem)):          # duyet tung hoc sinh
    ket_qua.append(diem[i] * 2 + 1) # tinh roi them vao cuoi
ket_qua = np.array(ket_qua)         # doi lai thanh mang NumPy
\end{lstlisting}
\textbf{Cách viết NumPy --- làm một lần cho cả mảng:}
\begin{lstlisting}
ket_qua = diem * 2 + 1
\end{lstlisting}
\begin{itemize}
  \item Cùng kết quả, 1 dòng thay cho 4 dòng, và nhanh hơn hàng chục lần
\end{itemize}
\end{frame}

\section{Broadcasting}

\begin{frame}[fragile]{Broadcasting --- khi hai mảng khác hình dạng}
\begin{lstlisting}
a = np.array([1, 2, 3])
print(a + 10)      # [11 12 13]  - so 10 duoc "phat" ra ca mang

diem = np.array([[8.0, 7.0, 9.0],     # 3 hoc sinh
                 [6.0, 8.5, 7.5],     # x 3 mon
                 [9.0, 9.0, 8.0]])
he_so = np.array([2, 1, 1])       # Toan he so 2, Ly 1, Hoa 1

print(diem * he_so)    # moi HANG deu duoc nhan voi he_so
\end{lstlisting}
\begin{block}{Broadcasting là gì?}
NumPy tự động "kéo giãn" mảng nhỏ hơn cho khớp hình dạng mảng lớn --- \textbf{không tốn thêm bộ nhớ}, không cần vòng lặp.
\end{block}
\end{frame}

\begin{frame}{Minh họa broadcasting}
\begin{center}
\begin{tikzpicture}[
    cell/.style={draw=npE, minimum width=0.62cm, minimum height=0.55cm, font=\scriptsize},
    ghost/.style={cell, fill=npWarn!15, dashed, draw=npWarn},

  ]
  % matrix 3x3
  \foreach \r in {0,1,2} \foreach \c in {0,1,2} {
    \node[cell, fill=npC!30] at (\c*0.62, -\r*0.55) {};
  }
  \node[font=\scriptsize\ttfamily] at (0.62, -2.05) {shape (3, 3)};

  \node[op] at (2.35, -0.55) {+};

  % vector shape (3,)
  \foreach \c/\v in {0/1, 1/2, 2/3} {
    \node[cell, fill=npB!45] at (3.1 + \c*0.62, 0) {\v};
  }
  \node[font=\scriptsize\ttfamily] at (3.72, -2.05) {shape (3,)};

  \node[op] at (5.45, -0.55) {=};

  % broadcast result
  \begin{scope}[xshift=6.2cm]
    \foreach \r in {0,1,2} \foreach \c/\v in {0/1, 1/2, 2/3} {

\end{tikzpicture}
\end{center}
\vspace{0.15cm}
\begin{itemize}
  \item Hàng \texttt{[1 2 3]} được lặp lại cho \textbf{mọi hàng} của ma trận
  \item Việc "kéo giãn" chỉ là \textbf{khái niệm} --- NumPy không thực sự tạo bản sao trong bộ nhớ
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Quy tắc broadcasting}
So sánh hình dạng \textbf{từ phải sang trái}. Hai chiều tương thích khi:
\begin{enumerate}
  \item Chúng \textbf{bằng nhau}, hoặc
  \item Một trong hai \textbf{bằng 1}, hoặc
  \item Một mảng \textbf{không có} chiều đó (được coi như bằng 1)
\end{enumerate}
\vspace{0.2cm}
\begin{center}
\small
\begin{tabular}{|l|l|l|}
\hline
\textbf{Hình dạng A} & \textbf{Hình dạng B} & \textbf{Kết quả} \\
\hline
\texttt{(3, 4)} & \texttt{(4,)}   & \texttt{(3, 4)} --- OK \\
\texttt{(3, 4)} & \texttt{(3, 1)} & \texttt{(3, 4)} --- OK \\
\texttt{(3, 1)} & \texttt{(1, 4)} & \texttt{(3, 4)} --- OK, giãn cả hai \\
\texttt{(3, 4)} & \texttt{(3,)}   & \textcolor{red}{Lỗi} --- 4 và 3 không khớp \\
\hline
\end{tabular}
\end{center}
\begin{lstlisting}
ValueError: operands could not be broadcast together
            with shapes (3,4) (3,)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]{Ứng dụng broadcasting: chuẩn hóa dữ liệu}
\textbf{Chuẩn hóa} (normalization): đưa các cột dữ liệu về cùng một thang đo, để cột có giá trị lớn không lấn át cột có giá trị nhỏ.
\begin{lstlisting}
# X: 100 mau du lieu, moi mau co 4 dac trung (features)
X = rng.normal(50, 10, size=(100, 4))

mean = X.mean(axis=0)   # shape (4,)  - trung binh TUNG COT
std  = X.std(axis=0)    # shape (4,)  - do lech chuan tung cot

X_chuan = (X - mean) / std     # broadcasting: (100,4) voi (4,)

print(X_chuan.mean(axis=0))    # ~ [0. 0. 0. 0.]
print(X_chuan.std(axis=0))     # ~ [1. 1. 1. 1.]
\end{lstlisting}
\begin{itemize}
  \item Chỉ \textbf{một dòng} thay cho hai vòng lặp lồng nhau và 400 phép tính thủ công
  \item Đây gọi là chuẩn hóa \textbf{z-score} --- công thức $z = (x - \mu) / \sigma$
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Đố vui: chọn axis nào?}
\begin{lstlisting}
# 4 hoc sinh (hang) x 3 mon: Toan, Ly, Hoa (cot)
diem = np.array([[8.0, 7.0, 9.0],
                 [6.0, 8.5, 7.5],
                 [9.0, 9.0, 8.0],
                 [5.0, 6.0, 6.5]])
\end{lstlisting}
\textbf{Viết một dòng lệnh cho mỗi yêu cầu:}
\begin{enumerate}
  \item Điểm trung bình của \textbf{từng học sinh} (ra 4 số)
  \pause
  \par\smallskip
  \hspace*{1em}$\to$ \texttt{diem.mean(axis=1)} \; \texttt{[8. 7.33 8.67 5.83]}
  \pause
  \item Điểm trung bình của \textbf{từng môn} (ra 3 số)
  \pause
  \par\smallskip
  \hspace*{1em}$\to$ \texttt{diem.mean(axis=0)} \; \texttt{[7. 7.625 7.75]}
  \pause
  \item Điểm \textbf{cao nhất} mà mỗi học sinh đạt được
  \pause
  \par\smallskip
  \hspace*{1em}$\to$ \texttt{diem.max(axis=1)} \; \texttt{[9. 8.5 9. 6.5]}
  \pause
  \item Mỗi học sinh có \textbf{mấy môn} từ 8 điểm trở lên?
  \pause
  \par\smallskip
  \hspace*{1em}$\to$ \texttt{(diem >= 8).sum(axis=1)} \; \texttt{[2 1 3 0]}
\end{enumerate}
\end{frame}

\begin{frame}[fragile]{Mẹo: giữ nguyên số chiều với keepdims}
\begin{lstlisting}
tb = diem.mean(axis=1)                  # shape (4,)
tb = diem.mean(axis=1, keepdims=True)   # shape (4, 1)

print(diem - tb)   # moi diem lech bao nhieu so voi DTB
                   # cua chinh hoc sinh do
\end{lstlisting}
\begin{itemize}
  \item \texttt{keepdims=True} giữ lại chiều vừa bị gộp $\to$ shape \texttt{(4,1)} thay vì \texttt{(4,)}
  \item Nhờ vậy \texttt{diem - tb} mới broadcasting đúng: mỗi \textbf{hàng} của \texttt{diem} trừ đi đúng giá trị trung bình của hàng đó
\end{itemize}
\end{frame}

\section{Biến đổi hình dạng và ghép mảng}

\begin{frame}[fragile]{reshape --- thay đổi hình dạng}
\begin{lstlisting}
a = np.arange(12)          # [0 1 2 ... 11]   shape (12,)

b = a.reshape(3, 4)        # 3 hang, 4 cot
c = a.reshape(4, 3)        # 4 hang, 3 cot
d = a.reshape(2, 2, 3)     # mang 3 chieu
e = a.reshape(3, -1)       # -1 = "tu tinh giup toi" -> (3, 4)

print(b.ravel())          # [0 1 ... 11]  tra ve 1 chieu (view)
print(b.flatten())         # giong ravel nhung tra ve BAN SAO
\end{lstlisting}
\begin{itemize}
  \item Số phần tử phải \textbf{không đổi}: $3 \times 4 = 12$ (được), còn $5 \times 3 = 15$ thì không
  \item \texttt{-1} nghĩa là "chiều này để NumPy tự tính": \texttt{a.reshape(3, -1)} $\to$ NumPy suy ra 4 cột
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Thêm chiều cho mảng}
\begin{lstlisting}
v = np.array([1, 2, 3])    # shape (3,) - khong hang, khong cot

v[:, np.newaxis]    # shape (3, 1)  - vector COT
v[np.newaxis, :]    # shape (1, 3)  - vector HANG
v.reshape(-1, 1)    # cach viet khac cho vector cot

a = np.array([[1, 2, 3],
              [4, 5, 6]])   # shape (2, 3)
a.T                 # chuyen vi -> shape (3, 2)  (xem phan sau)
\end{lstlisting}
\begin{block}{Điểm dễ nhầm}
Mảng \texttt{shape (3,)} \textbf{khác} \texttt{shape (3,1)} và \textbf{khác} \texttt{shape (1,3)}. Rất nhiều lỗi broadcasting đến từ chỗ này --- nhớ \texttt{print(x.shape)}!
\end{block}
\end{frame}

\begin{frame}[fragile]{Ghép và tách mảng}
\begin{lstlisting}
a = np.array([[1, 2], [3, 4]])
b = np.array([[5, 6], [7, 8]])

np.concatenate([a, b], axis=0)   # ghep theo HANG -> (4, 2)
np.concatenate([a, b], axis=1)   # ghep theo COT  -> (2, 4)
np.vstack([a, b])                # vertical   = axis 0
np.hstack([a, b])                # horizontal = axis 1
np.stack([a, b])                 # tao chieu MOI  -> (2, 2, 2)

x = np.arange(10)
np.split(x, 2)                   # tach lam 2 phan bang nhau
np.split(x, [3, 7])            # tach tai vi tri 3 va 7
\end{lstlisting}
\begin{itemize}
  \item \texttt{np.stack} thêm một chiều mới --- dùng khi gom nhiều ảnh cùng kích thước thành một chồng ảnh
\end{itemize}
\end{frame}

\section{Đại số tuyến tính với NumPy}

\begin{frame}[fragile]{Tích vô hướng của hai vector}
Hai vector \textbf{cùng độ dài}: nhân từng cặp phần tử tương ứng rồi cộng tất cả lại.
\[
  \vec{u} \cdot \vec{v} = u_1 v_1 + u_2 v_2 + u_3 v_3
  = 1{\times}4 + 2{\times}5 + 3{\times}6 = 32
\]
\begin{lstlisting}
u = np.array([1, 2, 3])
v = np.array([4, 5, 6])

np.dot(u, v)           # 32
(u * v).sum()          # 32 - nhan tung cap roi cong lai
np.linalg.norm(u)      # 3.74  do dai cua vector u
np.linalg.norm(u - v)  # 5.20  khoang cach giua u va v
\end{lstlisting}
\begin{itemize}
  \item \texttt{norm} là \textbf{độ dài} vector --- chính là định lý Pythagore mở rộng:
        $\sqrt{u_1^2 + u_2^2 + u_3^2} = \sqrt{1 + 4 + 9} \approx 3.74$
  \item \texttt{norm(u - v)} là \textbf{khoảng cách} giữa hai điểm --- dùng để đo hai dữ liệu giống nhau đến đâu
\end{itemize}
\end{frame}

\begin{frame}{Nhân ma trận được thực hiện như thế nào?}
\textbf{Quy tắc:} phần tử ở \textbf{hàng $i$, cột $j$} của kết quả
$=$ tích vô hướng của \textbf{hàng $i$} (ma trận trái) với \textbf{cột $j$} (ma trận phải).
\[
  \begin{pmatrix} \textcolor{npD}{\mathbf{1}} & \textcolor{npD}{\mathbf{2}} \\ 3 & 4 \end{pmatrix}
  \times
  \begin{pmatrix} \textcolor{npWarn}{\mathbf{5}} & 6 \\ \textcolor{npWarn}{\mathbf{7}} & 8 \end{pmatrix}
  =
  \begin{pmatrix} \mathbf{19} & 22 \\ 43 & 50 \end{pmatrix}
\]
\begin{center}
\small
$\mathbf{19} = \textcolor{npD}{1}{\times}\textcolor{npWarn}{5} + \textcolor{npD}{2}{\times}\textcolor{npWarn}{7}$
\qquad $22 = 1{\times}6 + 2{\times}8$
\qquad $43 = 3{\times}5 + 4{\times}7$
\qquad $50 = 3{\times}6 + 4{\times}8$
\end{center}
\begin{block}{Điều kiện về hình dạng}
\texttt{(m, n) @ (n, p)} $\to$ kết quả \texttt{(m, p)}.
Số \textbf{cột} của ma trận trái phải bằng số \textbf{hàng} của ma trận phải --- có vậy mới ghép cặp được từng phần tử.
\end{block}
\end{frame}

\begin{frame}[fragile]{Trong NumPy: \texttt{*} và \texttt{@} là hai phép khác nhau}
\begin{lstlisting}
A = np.array([[1, 2],
              [3, 4]])
B = np.array([[5, 6],
              [7, 8]])

A * B      # [[ 5 12], [21 32]]  nhan TUNG PHAN TU cung vi tri
A @ B      # [[19 22], [43 50]]  NHAN MA TRAN nhu tren
np.dot(A, B)     # giong A @ B
np.matmul(A, B)  # giong A @ B
\end{lstlisting}
\begin{alertblock}{Đừng nhầm!}
\texttt{*} nhân từng phần tử cùng vị trí (đòi hỏi hai mảng cùng hình dạng).
\texttt{@} là phép nhân ma trận trong toán học. Hai kết quả hoàn toàn khác nhau.
\end{alertblock}
\end{frame}

\begin{frame}{Chuyển vị: đổi hàng thành cột}
\textbf{Chuyển vị} (transpose) --- lật ma trận qua đường chéo chính: hàng thứ $i$ trở thành cột thứ $i$.
\[
  A = \begin{pmatrix} 1 & 2 & 3 \\ 4 & 5 & 6 \end{pmatrix}_{2 \times 3}
  \qquad
  A^{T} = \begin{pmatrix} 1 & 4 \\ 2 & 5 \\ 3 & 6 \end{pmatrix}_{3 \times 2}
\]
\begin{itemize}
  \item Phần tử ở vị trí (hàng $i$, cột $j$) chuyển sang vị trí (hàng $j$, cột $i$)
  \item Hình dạng \texttt{(m, n)} trở thành \texttt{(n, m)}
  \item Trong NumPy chỉ cần viết \texttt{A.T} --- và đây là một \textbf{view}, không tốn thêm bộ nhớ
  \item Hay dùng khi hai mảng "lệch chiều" nhau, ví dụ muốn nhân \texttt{(2,3)} với \texttt{(2,3)}:
        không nhân được, nhưng \texttt{A @ B.T} thì được
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Vì sao hệ phương trình viết được dưới dạng ma trận?}
\[
  \left\{
  \begin{array}{r@{\;}c@{\;}l}
    2x + \phantom{3}y &=& 5 \\
    \phantom{2}x + 3y &=& 10
  \end{array}
  \right.
  \qquad\Longleftrightarrow\qquad
  \underbrace{\begin{pmatrix} 2 & 1 \\ 1 & 3 \end{pmatrix}}_{A}
  \underbrace{\begin{pmatrix} x \\ y \end{pmatrix}}_{\vec{x}}
  =
  \underbrace{\begin{pmatrix} 5 \\ 10 \end{pmatrix}}_{\vec{b}}
\]
Nhân \textbf{hàng 1} của $A$ với vector ẩn theo đúng quy tắc ở slide trước:
$2 {\times} x + 1 {\times} y$ --- chính là vế trái của phương trình thứ nhất.
Tương tự hàng 2 cho ra $x + 3y$.
\begin{lstlisting}
A = np.array([[2, 1], [1, 3]])
b = np.array([5, 10])

np.linalg.solve(A, b)     # [1. 3.]  -> x = 1, y = 3
A @ np.array([1, 3])      # [ 5 10]  kiem tra lai: dung!
\end{lstlisting}
\end{frame}

\section{NumPy trong các bài toán AI}

\begin{frame}[fragile]{Ảnh số chính là một mảng NumPy}
\begin{lstlisting}
# Anh mau: (cao, rong, 3 kenh RGB), kieu uint8 gia tri 0..255
img = rng.integers(0, 256, size=(128, 128, 3), dtype=np.uint8)

img.shape          # (128, 128, 3)
img[0, 0]          # [R G B] cua diem anh goc tren trai

img_gray = img.mean(axis=2)        # anh xam: trung binh 3 kenh
img_flip = img[:, ::-1, :]         # lat anh trai-phai
img_crop = img[32:96, 32:96]       # cat vung giua
img_norm = img.astype(np.float32) / 255.0  # dua ve [0, 1]

do_sang = img.mean()               # do sang trung binh
kenh_do = img[:, :, 0]             # tach rieng kenh do
\end{lstlisting}
\begin{itemize}
  \item Lật ảnh, cắt ảnh, đổi độ sáng --- các thao tác xử lý ảnh quen thuộc --- chỉ là slicing và phép toán trên mảng
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Vài hàm số hay gặp khi học AI}
\[
  \mathrm{sigmoid}(x) = \frac{1}{1 + e^{-x}}
  \qquad
  \mathrm{ReLU}(x) = \max(0,\; x)
  \qquad
  \tanh(x) = \frac{e^{x} - e^{-x}}{e^{x} + e^{-x}}
\]
\begin{lstlisting}
def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def relu(x):
    return np.maximum(0, x)

x = np.linspace(-5, 5, 11)
print(np.round(sigmoid(x), 2))  # [0.01 0.02 ... 0.98 0.99]
print(relu(x))                  # [0. 0. ... 4. 5.]
\end{lstlisting}
\begin{itemize}
  \item \texttt{sigmoid} nén \textbf{mọi} số thực về khoảng $(0, 1)$
  \item \texttt{ReLU} giữ nguyên số dương, cắt số âm về $0$
  \item Viết được cho \textbf{cả mảng} chỉ nhờ vector hóa --- không cần vòng lặp
\end{itemize}
\end{frame}

\begin{frame}[fragile]{softmax --- biến dãy số thành tỉ lệ phần trăm}
\[
  \mathrm{softmax}(x)_i = \frac{e^{x_i}}{e^{x_1} + e^{x_2} + \dots + e^{x_n}}
\]
\begin{lstlisting}
def softmax(x):
    e = np.exp(x - np.max(x))   # tru max de tranh tran so
    return e / np.sum(e)

diem_so = np.array([2.0, 1.0, 0.1])
print(softmax(diem_so))        # [0.659 0.242 0.099] - tong = 1
\end{lstlisting}
\begin{itemize}
  \item Mọi kết quả đều \textbf{dương} và \textbf{tổng bằng 1} $\to$ đọc được như tỉ lệ phần trăm:
        $65.9\%$, $24.2\%$, $9.9\%$
  \item Số nào lớn hơn thì tỉ lệ cao hơn --- nhưng chênh lệch bị khuếch đại (do hàm $e^{x}$)
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Đo sai số giữa hai dãy số}
\textbf{Ví dụ:} nhiệt độ \textit{dự báo} và nhiệt độ \textit{thực tế} của 4 ngày --- dự báo lệch trung bình bao nhiêu độ?
\begin{lstlisting}
du_bao   = np.array([28.0, 31.0, 27.5, 33.0])
thuc_te  = np.array([27.8, 31.2, 29.0, 32.5])

sai_so = du_bao - thuc_te        # [0.2 -0.2 -1.5 0.5]

# Sai so tuyet doi trung binh (MAE)
np.mean(np.abs(sai_so))          # 0.6 do

# Sai so binh phuong trung binh (MSE) - phat nang sai so lon
np.mean(sai_so ** 2)             # 0.645
\end{lstlisting}
\textbf{Đếm tỉ lệ giống nhau giữa hai dãy:}
\begin{lstlisting}
dap_an  = np.array([1, 3, 2, 4, 1])   # dap an dung
bai_lam = np.array([1, 3, 4, 4, 2])   # bai lam cua hoc sinh
np.mean(dap_an == bai_lam)            # 0.6 -> lam dung 60%
\end{lstlisting}
\end{frame}
